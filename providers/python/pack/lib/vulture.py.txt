def exact_object(value, keys, name):
    if not isinstance(value, dict) or set(value) != set(keys):
        raise ValueError(f"{name} has invalid fields")


def bounded(value):
    return value.encode("utf-8")[:MAX_STRING_BYTES].decode("utf-8", "ignore")


def exact_path(value):
    if (
        not isinstance(value, str)
        or not value
        or len(value.encode("utf-8")) > MAX_STRING_BYTES
        or value.startswith("/")
        or "\\" in value
        or any(part in {"", ".", ".."} for part in value.split("/"))
        or not value.endswith((".py", ".pyi"))
    ):
        raise ValueError("request contains an invalid Python path")
    return value


def identifier_chain(value):
    return (
        isinstance(value, str)
        and value
        and all(
            part.isascii() and part.isidentifier() and not part[0].isdigit()
            for part in value.split(".")
        )
    )


def contained_file(root, path):
    target = os.path.realpath(os.path.join(root, *path.split("/")))
    if os.path.commonpath((root, target)) != root or not os.path.isfile(target):
        raise ValueError(f"Python source {path} escapes the project")
    return target


def definition_location(path, node, name):
    decorators = getattr(node, "decorator_list", ())
    line = decorators[0].lineno if decorators else node.lineno
    if isinstance(node, (ast.Assign, ast.AnnAssign, ast.Name, ast.Attribute)):
        end_line = line
    else:
        end_line = node.end_lineno
    return path, line, end_line, name


def assignment_targets(node):
    if isinstance(node, (ast.Tuple, ast.List)):
        for item in node.elts:
            yield from assignment_targets(item)
    elif isinstance(node, ast.Name):
        yield node


def definition_sites(file, body, name):
    result = []
    for node in body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            if node.name == name:
                result.append(("definition", node))
        elif isinstance(node, (ast.Assign, ast.AnnAssign)):
            targets = node.targets if isinstance(node, ast.Assign) else [node.target]
            for target in targets:
                for assigned in assignment_targets(target):
                    if assigned.id == name:
                        result.append(("definition", assigned))
        elif isinstance(node, ast.ImportFrom):
            for item in node.names:
                if (item.asname or item.name.partition(".")[0]) == name:
                    result.append(("import", node, item))
    return result


def module_file(modules, module):
    candidates = modules.get(module, [])
    runtime = [file for file in candidates if file["path"].endswith(".py")]
    selected = runtime if runtime else candidates
    locations = {
        file["path"].removesuffix(".py").removesuffix(".pyi") for file in selected
    }
    if not selected:
        raise ValueError("module has no project definition")
    if len(locations) != 1 or len(selected) != 1:
        raise ValueError("module is ambiguous")
    return selected[0]


def class_member(file, node, parts):
    result = [definition_location(file["path"], node, parts[0])]
    if len(parts) == 1:
        return result
    if not isinstance(node, ast.ClassDef):
        raise TypeError("symbol has no class member")
    found = definition_sites(file, node.body, parts[1])
    if len(found) != 1 or found[0][0] != "definition":
        raise ValueError("class member is stale or ambiguous")
    return result + class_member(file, found[0][1], parts[1:])


def module_reference(modules, module, parts, seen):
    identity = module, ".".join(parts)
    if identity in seen:
        raise ValueError("reference re-export cycle")
    file = module_file(modules, module)
    found = definition_sites(file, file["tree"].body, parts[0])
    if len(found) != 1:
        raise ValueError("symbol is stale or ambiguous")
    match = found[0]
    if match[0] == "import":
        node, item = match[1:]
        target_module = import_from_module(file, node)
        if not target_module:
            raise ValueError("re-export has no module")
        location = definition_location(
            file["path"], node, item.asname or item.name.partition(".")[0]
        )
        return [location] + module_reference(
            modules,
            target_module,
            item.name.split(".") + parts[1:],
            seen | {identity},
        )
    return class_member(file, match[1], parts)


class ContainedAstroidReferences:
    def __init__(self, files, contracts):
        from astroid import AstroidError, AstroidImportError
        from astroid.builder import AstroidBuilder
        from astroid.manager import AstroidManager

        self.manager_type = AstroidManager
        self.import_error = AstroidImportError
        self.errors = (AstroidError, RecursionError)
        self.sources = {file["path"]: file for file in files}
        self.modules = {}
        for file in files:
            if file["module"] and file["path"].endswith(".py"):
                self.modules.setdefault(file["module"], []).append(file)
        self.module_roots = {name.split(".")[0] for name in self.modules}
        self.manager = AstroidManager()
        self.builder = AstroidBuilder(self.manager, apply_transforms=False)
        self.cache = {}
        self.paths = {}
        self.expression_indexes = {}
        self.declared_roots = {
            contract["target"] for contract in contracts if contract["kind"] == "type"
        }
        model_names = defaultdict(set)
        for target in self.declared_roots:
            module, separator, name = target.rpartition(".")
            if separator:
                model_names[module].add(name)
        self.models = {
            module: "\n".join(f"class {name}: pass" for name in sorted(names))
            for module, names in model_names.items()
        }
        self.override = patch.object(
            self.manager_type, "ast_from_module_name", self.load
        )

    def __enter__(self):
        self.override.__enter__()
        return self

    def __exit__(self, kind, value, traceback):
        return self.override.__exit__(kind, value, traceback)

    def load(self, name, context_file=None, use_cache=True):
        if context_file is not None and context_file not in self.sources:
            raise self.import_error("contract inference requested external source")
        if use_cache and name in self.cache:
            return self.cache[name]
        candidates = self.modules.get(name, [])
        if len(candidates) == 1:
            source = candidates[0]
            module = self.builder.string_build(source["source"], name, source["path"])
            module.package = (
                source["path"].endswith("/__init__.py")
                or source["path"] == "__init__.py"
            )
            self.paths[name] = source["path"]
        elif candidates or name.split(".")[0] in self.module_roots:
            raise self.import_error(f"unresolved contained contract module: {name}")
        elif name in self.models:
            module = self.builder.string_build(self.models[name], name)
        elif any(model.startswith(name + ".") for model in self.models):
            module = self.builder.string_build("", name)
            module.package = True
        else:
            raise self.import_error(f"unadmitted contract module: {name}")
        self.cache[name] = module
        return module

    def entry_point(self, contract):
        try:
            return self.resolve_entry_point(contract)
        except self.errors as error:
            raise ValueError(str(error)) from error

    def type_classes(self, contract):
        try:
            return self.resolve_type_contract(contract)
        except self.errors as error:
            raise ValueError(str(error)) from error

    def resolve_type_contract(self, contract):
        from astroid import nodes

        for name in sorted(self.modules):
            self.load(name)
        roots = {contract["target"]}
        classes = set()
        for name, path in sorted(self.paths.items()):
            module = self.cache[name]
            for node in module.nodes_of_class(nodes.ClassDef):
                if self.has_base(node, roots, set()):
                    classes.add((path, node.lineno, node.name))
        return classes

    def has_base(self, node, roots, seen):
        from astroid import nodes

        if node in seen or len(seen) >= MAX_ASTROID_DEPTH:
            return False
        seen = seen | {node}
        if node.qname() in roots & self.declared_roots:
            return True
        for base in node.bases:
            names = list(base.nodes_of_class(nodes.Name))
            if any(not self.unique_binding(name) for name in names):
                continue
            inferred = list(base.infer())
            if len(inferred) != 1 or not isinstance(inferred[0], nodes.ClassDef):
                continue
            parent = inferred[0]
            if parent.root().name not in self.paths and parent.qname() in roots:
                return True
            if self.has_base(parent, roots, seen):
                return True
        return False

    def unique_binding(self, node):
        definitions = node.lookup(node.name)[1]
        return len(definitions) == 1 and (
            definitions[0].root() is not node.root()
            or definitions[0].lineno <= node.lineno
        )

    def expression_applies(self, file, node, contract, applicable):
        from astroid import nodes

        try:
            expression = self.astroid_expression(file, node)
            if expression is None:
                return False
            if isinstance(expression, nodes.Const) and isinstance(
                expression.value, str
            ):
                inferred = self.string_reference(expression)
            else:
                names = expression.nodes_of_class(nodes.Name)
                if any(not self.unique_binding(name) for name in names):
                    return False
                inferred = self.unique_inference(expression)
            return self.inferred_applies(inferred, contract, applicable)
        except (*self.errors, AttributeError, TypeError, ValueError):
            return False

    def astroid_expression(self, file, node):
        from astroid import nodes

        module = self.cache.get(file["module"])
        labels = {ast.Name: "name", ast.Attribute: "attribute", ast.Constant: "text"}
        label = labels.get(type(node))
        if module is None or label is None:
            return None
        if file["module"] not in self.expression_indexes:
            index = defaultdict(list)
            for candidate in module.nodes_of_class(
                (nodes.Name, nodes.Attribute, nodes.Const)
            ):
                if isinstance(candidate, nodes.Name):
                    key = (
                        "name",
                        candidate.fromlineno,
                        candidate.col_offset,
                        candidate.name,
                    )
                elif isinstance(candidate, nodes.Attribute):
                    key = (
                        "attribute",
                        candidate.fromlineno,
                        candidate.col_offset,
                        candidate.as_string(),
                    )
                elif isinstance(candidate.value, str):
                    key = (
                        "text",
                        candidate.fromlineno,
                        candidate.col_offset,
                        candidate.value,
                    )
                else:
                    continue
                index[key].append(candidate)
            self.expression_indexes[file["module"]] = index
        rendered = node.value if isinstance(node, ast.Constant) else ast.unparse(node)
        key = label, node.lineno, node.col_offset, rendered
        matches = self.expression_indexes[file["module"]].get(key, [])
        return matches[0] if len(matches) == 1 else None

    def unique_inference(self, node):
        values = []
        for value in node.infer():
            if value not in values:
                values.append(value)
            if len(values) > 1:
                return None
        return values[0] if len(values) == 1 else None

    def string_reference(self, expression):
        from astroid import nodes

        if not identifier_chain(expression.value):
            return None
        parts = expression.value.split(".")
        definitions = expression.scope().lookup(parts[0])[1]
        if len(definitions) != 1:
            return None
        definition = definitions[0]
        if (
            definition.root() is expression.root()
            and definition.fromlineno > expression.fromlineno
        ):
            return None
        if isinstance(definition, nodes.ImportFrom):
            names = [
                original
                for original, alias in definition.names
                if (alias or original) == parts[0]
            ]
            if len(names) != 1:
                return None
            imported = definition.do_import_module(definition.modname)
            values = imported.getattr(names[0])
            current = self.unique_inference(values[0]) if len(values) == 1 else None
        elif isinstance(definition, nodes.Import):
            names = [
                imported
                for imported, alias in definition.names
                if (alias or imported.partition(".")[0]) == parts[0]
            ]
            if len(names) != 1:
                return None
            module_name = (
                names[0]
                if any(alias == parts[0] for _, alias in definition.names)
                else names[0].partition(".")[0]
            )
            current = definition.do_import_module(module_name)
        else:
            current = self.unique_inference(definition)
        for part in parts[1:]:
            if current is None:
                return None
            values = current.getattr(part)
            current = self.unique_inference(values[0]) if len(values) == 1 else None
        return current

    def inferred_applies(self, inferred, contract, applicable):
        from astroid import nodes

        if not isinstance(inferred, nodes.ClassDef):
            return False
        path = self.paths.get(inferred.root().name)
        if path is None:
            return inferred.qname() == contract["target"]
        return (path, inferred.lineno, inferred.name) in applicable

    def resolve_entry_point(self, contract):
        module_name, symbol = contract["target"].split(":", 1)
        module = self.load(module_name)
        if module_name not in self.paths:
            raise ValueError(f"entry-point module is not project source: {module_name}")
        current = module
        kept = set()
        for part in symbol.split("."):
            current = self.member(current, part, kept)
        for name in contract["members"]:
            self.member(current, name, kept)
        return kept

    def member(self, owner, name, kept):
        from astroid import bases, nodes

        definitions = owner.getattr(name)
        if len(definitions) != 1:
            raise ValueError(f"entry-point member is stale or ambiguous: {name}")
        definition = definitions[0]
        self.unconditional(definition)
        path = self.paths.get(definition.root().name)
        if path is None:
            raise ValueError(f"entry-point member is not project source: {name}")
        start = definition.fromlineno
        end = (
            definition.tolineno
            if isinstance(definition, (nodes.FunctionDef, nodes.ClassDef))
            else start
        )
        kept.add((path, start, end, name))
        inferred = []
        for value in definition.infer():
            if value not in inferred:
                inferred.append(value)
            if len(inferred) > 1:
                break
        if len(inferred) != 1 or not isinstance(
            inferred[0],
            (nodes.Module, nodes.ClassDef, nodes.FunctionDef, bases.Instance),
        ):
            raise ValueError(f"entry-point member cannot resolve uniquely: {name}")
        return inferred[0]

    def unconditional(self, definition):
        from astroid import nodes

        current = definition.parent
        while current is not None:
            if isinstance(
                current, (nodes.If, nodes.For, nodes.While, nodes.Try, nodes.With)
            ):
                raise TypeError("entry-point definition is conditional")
            if isinstance(current, (nodes.FunctionDef, nodes.ClassDef, nodes.Module)):
                return
            current = current.parent


def import_from_module(file, node):
    parts = []
    if node.level:
        package = file["package"]
        if not package:
            return ""
        parts = package.split(".")
        if len(parts) < node.level - 1:
            return ""
        parts = parts[: len(parts) - node.level + 1]
    if node.module:
        parts.append(node.module)
    return ".".join(parts)


def import_bindings(file):
    result = {}
    for node in file["tree"].body:
        if isinstance(node, ast.Import):
            for item in node.names:
                result[item.asname or item.name.partition(".")[0]] = (
                    item.name if item.asname else item.name.partition(".")[0]
                )
        elif isinstance(node, ast.ImportFrom):
            module = import_from_module(file, node)
            if module:
                for item in node.names:
                    if item.name != "*":
                        result[item.asname or item.name] = module + "." + item.name
    return result


def expanded_name(node, bindings):
    if isinstance(node, ast.Name):
        return bindings.get(node.id, node.id)
    if isinstance(node, ast.Attribute):
        parent = expanded_name(node.value, bindings)
        return parent + "." + node.attr if parent else ""
    return ""


def statement_binding_names(node):
    if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
        return {node.name}
    if isinstance(node, ast.Lambda):
        return set()
    if isinstance(node, (ast.Import, ast.ImportFrom)):
        return {
            item.asname or item.name.partition(".")[0]
            for item in node.names
            if item.name != "*"
        }
    result = set()
    if isinstance(node, ast.Name) and isinstance(node.ctx, ast.Store):
        result.add(node.id)
    if isinstance(node, ast.ExceptHandler) and node.name:
        result.add(node.name)
    for child in ast.iter_child_nodes(node):
        result.update(statement_binding_names(child))
    return result


def exact_import_bindings(file):
    writes = defaultdict(int)
    candidates = defaultdict(list)
    for node in file["tree"].body:
        for name in statement_binding_names(node):
            writes[name] += 1
        if isinstance(node, ast.Import):
            for item in node.names:
                name = item.asname or item.name.partition(".")[0]
                target = item.name if item.asname else item.name.partition(".")[0]
                candidates[name].append(target)
        elif isinstance(node, ast.ImportFrom):
            module = import_from_module(file, node)
            if module:
                for item in node.names:
                    if item.name != "*":
                        candidates[item.asname or item.name].append(
                            module + "." + item.name
                        )
    return {
        name: values[0]
        for name, values in candidates.items()
        if len(values) == 1 and writes[name] == 1
    }


def root_name(node):
    while isinstance(node, ast.Attribute):
        node = node.value
    return node.id if isinstance(node, ast.Name) else ""


def decorator_target_matches(node, expected, bindings, blocked):
    target = node.func if isinstance(node, ast.Call) else node
    root = root_name(target)
    if not root or root in blocked or root not in bindings:
        return False
    return expanded_name(target, bindings) == expected


def decorator_matches(node, contract, bindings, blocked):
    if not decorator_target_matches(node, contract["target"], bindings, blocked):
        return False
    keywords = contract["keywords"]
    if not keywords:
        return True
    if not isinstance(node, ast.Call) or node.args:
        return False
    actual = {keyword.arg: keyword.value for keyword in node.keywords}
    return None not in actual and all(
        name in actual
        and isinstance(actual[name], ast.Constant)
        and actual[name].value is value
        for name, value in keywords.items()
    )


def decorator_body_locations(file, body, contract, bindings, blocked):
    result = set()
    for node in body:
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            continue
        if any(
            decorator_matches(decorator, contract, bindings, blocked)
            for decorator in node.decorator_list
        ):
            result.add(definition_location(file["path"], node, node.name))
        if isinstance(node, ast.ClassDef):
            local = set()
            for statement in node.body:
                local.update(statement_binding_names(statement))
            result.update(
                decorator_body_locations(
                    file, node.body, contract, bindings, blocked | local
                )
            )
    return result


def decorator_locations(files, contract):
    result = set()
    for file in files:
        bindings = exact_import_bindings(file)
        result.update(
            decorator_body_locations(file, file["tree"].body, contract, bindings, set())
        )
    return result


def class_scope_nodes(node):
    yield node
    if isinstance(
        node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)
    ):
        return
    for child in ast.iter_child_nodes(node):
        yield from class_scope_nodes(child)


def class_var_annotation(annotation, bindings):
    if not isinstance(annotation, ast.Subscript):
        return False
    root = root_name(annotation.value)
    if not root or root not in bindings:
        return False
    return expanded_name(annotation.value, bindings) in {
        "typing.ClassVar",
        "typing_extensions.ClassVar",
    }


def reference_name(node):
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.Attribute):
        parent = reference_name(node.value)
        return parent + "." + node.attr if parent else ""
    return ""


def connection_path(node):
    name = reference_name(node)
    return tuple(name.split(".")) if name else ()


def source_write_counts(file):
    return Counter(
        (
            node.lineno,
            node.attr if isinstance(node, ast.Attribute) else node.id,
        )
        for node in ast.walk(file["tree"])
        if isinstance(node, (ast.Attribute, ast.Name))
        and isinstance(node.ctx, ast.Store)
    )


class ConnectionFlow(ast.NodeVisitor):
    def __init__(self, connection_factory):
        self.connections = set()
        self.mutable_names = set()
        self.scope = "module"
        self.connection_factory = connection_factory

    def connection(self, node):
        if isinstance(node, ast.Call):
            return self.connection_factory(node)
        return connection_path(node) in self.connections

    def bind(self, target, connected=False):
        if isinstance(target, (ast.Tuple, ast.List)):
            for item in target.elts:
                self.bind(item)
            return
        path = connection_path(target)
        self.connections = {
            known
            for known in self.connections
            if known[: len(path)] != path
            and not (len(path) > 1 and path[-1] in known[1:])
        }
        if connected and path:
            self.connections.add(path)

    def visit_Assign(self, node):
        connected = self.connection(node.value)
        self.visit(node.value)
        for target in node.targets:
            self.visit(target)
            self.bind(target, connected)

    def visit_AnnAssign(self, node):
        if node.value is None:
            return
        connected = self.connection(node.value)
        self.visit(node.value)
        self.visit(node.target)
        self.bind(node.target, connected)

    def visit_AugAssign(self, node):
        self.generic_visit(node)
        self.bind(node.target)

    def visit_NamedExpr(self, node):
        connected = self.connection(node.value)
        self.visit(node.value)
        self.bind(node.target, connected)

    def visit_Delete(self, node):
        for target in node.targets:
            self.bind(target)

    def visit_Call(self, node):
        self.generic_visit(node)
        self.connections = {
            path
            for path in self.connections
            if len(path) == 1
            and path[0] not in self.mutable_names
            and self.scope != "module"
        }

    def block(self, statements, initial):
        self.connections = initial.copy()
        for statement in statements:
            self.visit(statement)
        return self.connections.copy()

    def visit_If(self, node):
        self.visit(node.test)
        initial = self.connections.copy()
        left = self.block(node.body, initial)
        right = self.block(node.orelse, initial)
        self.connections = left & right

    def visit_IfExp(self, node):
        self.visit(node.test)
        initial = self.connections.copy()
        left = self.block([node.body], initial)
        right = self.block([node.orelse], initial)
        self.connections = left & right

    def visit_BoolOp(self, node):
        possible = []
        for value in node.values:
            self.visit(value)
            possible.append(self.connections.copy())
        self.connections = set.intersection(*possible)

    def visit_Try(self, node):
        normal = self.block(node.body, self.connections)
        normal = self.block(node.orelse, normal)
        exits = [normal]
        for handler in node.handlers:
            exits.append(self.block(handler.body, set()))
        merged = set.intersection(*exits)
        self.connections = (
            self.block(node.finalbody, set()) if node.finalbody else merged
        )

    def visit_TryStar(self, node):
        self.visit_Try(node)

    def visit_For(self, node):
        self.visit(node.iter)
        self.loop(node)

    visit_AsyncFor = visit_For

    def visit_While(self, node):
        self.visit(node.test)
        self.loop(node)

    def loop(self, node):
        initial = self.connections.copy()
        body = self.block(node.body, set())
        self.connections = self.block(node.orelse, initial & body)

    def visit_With(self, node):
        initial = self.connections.copy()
        nonsuppressing = True
        for item in node.items:
            connected = self.connection(item.context_expr)
            nonsuppressing = nonsuppressing and connected
            self.visit(item.context_expr)
            if item.optional_vars is not None:
                self.bind(item.optional_vars, connected)
        normal = self.block(node.body, self.connections)
        self.connections = normal if nonsuppressing else initial & normal

    def visit_AsyncWith(self, node):
        self.block(node.body, set())
        self.connections.clear()

    def visit_Import(self, node):
        for alias in node.names:
            self.bind(ast.Name(id=alias.asname or alias.name.split(".")[0]))

    def visit_ImportFrom(self, node):
        for alias in node.names:
            if alias.name == "*":
                self.connections.clear()
            else:
                self.bind(ast.Name(id=alias.asname or alias.name))

    def visit_Match(self, node):
        self.visit(node.subject)
        initial = self.connections.copy()
        exits = [initial]
        for case in node.cases:
            exits.append(self.block(case.body, set()))
        self.connections = set.intersection(*exits)

    def visit_Lambda(self, node):
        return

    def visit_ListComp(self, node):
        self.connections.clear()

    visit_SetComp = visit_ListComp
    visit_DictComp = visit_ListComp
    visit_GeneratorExp = visit_ListComp


class TypeAttributeVisitor(ConnectionFlow):
    def __init__(self, file, contract, resolver, applicable, writes):
        super().__init__(self.connection_call)
        self.file = file
        self.contract = contract
        self.resolver = resolver
        self.applicable = applicable
        self.attributes = set(contract["attributes"])
        self.writes = writes
        self.owner = False
        self.kept = set()

    def keep(self, node):
        if self.writes[(node.lineno, node.attr)] <= 1:
            self.kept.add(definition_location(self.file["path"], node, node.attr))

    def connection_call(self, node):
        return self.resolver.expression_applies(
            self.file, node.func, self.contract, self.applicable
        )

    def parameters(self, node):
        result = set()
        arguments = node.args.posonlyargs + node.args.args + node.args.kwonlyargs
        for argument in arguments:
            if argument.annotation is not None and self.resolver.expression_applies(
                self.file, argument.annotation, self.contract, self.applicable
            ):
                result.add((argument.arg,))
        decorators = {reference_name(value) for value in node.decorator_list}
        if (
            arguments
            and self.owner
            and not decorators & {"staticmethod", "classmethod"}
        ):
            result.add((arguments[0].arg,))
        return result

    def visit_ClassDef(self, node):
        previous = self.scope, self.owner, self.connections, self.mutable_names
        self.owner = (self.file["path"], node.lineno, node.name) in self.applicable
        self.scope = f"{self.scope}/class:{node.lineno}:{node.col_offset + 1}"
        self.connections = self.block(node.body, set())
        self.scope, self.owner, self.connections, self.mutable_names = previous
        self.bind(ast.Name(id=node.name))
        self.connections = {path for path in self.connections if len(path) == 1}

    def visit_FunctionDef(self, node):
        previous = self.scope, self.owner, self.connections, self.mutable_names
        self.scope = f"{self.scope}/function:{node.lineno}:{node.col_offset + 1}"
        self.connections = self.parameters(node)
        self.owner = False
        self.mutable_names = {
            name
            for value in ast.walk(node)
            if isinstance(value, (ast.Nonlocal, ast.Global))
            for name in value.names
        }
        for statement in node.body:
            self.visit(statement)
        self.scope, self.owner, self.connections, self.mutable_names = previous
        self.bind(ast.Name(id=node.name))
        self.connections = {path for path in self.connections if len(path) == 1}

    visit_AsyncFunctionDef = visit_FunctionDef

    def visit_Attribute(self, node):
        if (
            isinstance(node.ctx, ast.Store)
            and node.attr in self.attributes
            and self.connection(node.value)
        ):
            self.keep(node)
        self.generic_visit(node)


def type_contract_locations(files, contract, resolver):
    applicable = resolver.type_classes(contract)
    members = set(contract["members"])
    attributes = set(contract["attributes"])
    decorators = set(contract["decorators"])
    result = set()
    for file in files:
        writes = source_write_counts(file)
        classes = {
            (line, name) for path, line, name in applicable if path == file["path"]
        }
        bindings = exact_import_bindings(file)
        for node in ast.walk(file["tree"]):
            if (
                not isinstance(node, ast.ClassDef)
                or (node.lineno, node.name) not in classes
            ):
                continue
            blocked = set()
            for statement in node.body:
                blocked.update(statement_binding_names(statement))
            for statement in node.body:
                for item in class_scope_nodes(statement):
                    if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        decorated = any(
                            decorator_target_matches(
                                decorator, expected, bindings, blocked
                            )
                            for decorator in item.decorator_list
                            for expected in decorators
                        )
                        if item.name in members or decorated:
                            result.add(
                                definition_location(file["path"], item, item.name)
                            )
                    elif isinstance(item, ast.Assign):
                        for target in item.targets:
                            if (
                                isinstance(target, ast.Name)
                                and target.id in attributes
                                and writes[(target.lineno, target.id)] <= 1
                            ):
                                result.add(
                                    definition_location(file["path"], target, target.id)
                                )
                    elif (
                        isinstance(item, ast.AnnAssign)
                        and isinstance(item.target, ast.Name)
                        and (
                            item.target.id in attributes
                            or contract["annotatedFields"]
                            and not class_var_annotation(item.annotation, bindings)
                        )
                        and writes[(item.target.lineno, item.target.id)] <= 1
                    ):
                        result.add(
                            definition_location(
                                file["path"], item.target, item.target.id
                            )
                        )
        if attributes:
            visitor = TypeAttributeVisitor(file, contract, resolver, applicable, writes)
            visitor.visit(file["tree"])
            result.update(visitor.kept)
    return result


def module_scope_nodes(node):
    if isinstance(
        node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)
    ):
        return
    yield node
    for child in ast.iter_child_nodes(node):
        yield from module_scope_nodes(child)


def module_binding_value(node, contract, bindings):
    if isinstance(node, (ast.List, ast.Tuple)):
        return bool(node.elts) and all(
            module_binding_value(value, contract, bindings) for value in node.elts
        )
    target = node.func if isinstance(node, ast.Call) else node
    root = root_name(target)
    if not root or root not in bindings:
        return False
    qualified = expanded_name(target, bindings)
    return qualified.startswith(contract["target"] + ".")


def module_binding_locations(files, contract):
    result = set()
    members = set(contract["members"])
    for file in files:
        bindings = exact_import_bindings(file)
        for node in module_scope_nodes(file["tree"]):
            if isinstance(node, ast.Assign):
                targets = node.targets
            elif isinstance(node, ast.AnnAssign):
                targets = [node.target]
            else:
                continue
            if not module_binding_value(node.value, contract, bindings):
                continue
            for target in targets:
                if isinstance(target, ast.Name) and target.id in members:
                    result.add(definition_location(file["path"], target, target.id))
    return result


def callback_locations(file):
    result = set()
    bindings = import_bindings(file)
    zip_infos = set()
    protocols = {
        "ast.NodeVisitor": lambda name: name.startswith("visit_"),
        "ast.NodeTransformer": lambda name: name.startswith("visit_"),
        "urllib.request.BaseHandler": lambda name: (
            name in {"redirect_request", "default_open", "unknown_open", "proxy_open"}
            or name.startswith("http_error_")
            or name.endswith("_open")
        ),
        "urllib.request.HTTPRedirectHandler": lambda name: (
            name == "redirect_request" or name.startswith("http_error_")
        ),
        "html.parser.HTMLParser": lambda name: (
            name
            in {
                "handle_starttag",
                "handle_startendtag",
                "handle_endtag",
                "handle_data",
                "handle_entityref",
                "handle_charref",
                "handle_comment",
                "handle_decl",
                "unknown_decl",
            }
        ),
    }
    for node in ast.walk(file["tree"]):
        if isinstance(node, ast.ClassDef):
            predicates = [
                protocols.get(expanded_name(base, bindings)) for base in node.bases
            ]
            predicates = [
                predicate for predicate in predicates if predicate is not None
            ]
            for child in node.body:
                if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)) and any(
                    predicate(child.name) for predicate in predicates
                ):
                    result.add(definition_location(file["path"], child, child.name))
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name in {
            "__exit__",
            "__aexit__",
        }:
            arguments = (
                list(node.args.posonlyargs)
                + list(node.args.args)
                + list(node.args.kwonlyargs)
            )
            if node.args.vararg is not None:
                arguments.append(node.args.vararg)
            if node.args.kwarg is not None:
                arguments.append(node.args.kwarg)
            for argument in arguments:
                result.add(definition_location(file["path"], argument, argument.arg))
        if isinstance(node, (ast.Assign, ast.AnnAssign)):
            value = node.value
            if (
                isinstance(value, ast.Call)
                and expanded_name(value.func, bindings) == "zipfile.ZipInfo"
            ):
                targets = (
                    node.targets if isinstance(node, ast.Assign) else [node.target]
                )
                for target in targets:
                    if isinstance(target, ast.Name):
                        zip_infos.add(target.id)
        if (
            isinstance(node, ast.Attribute)
            and isinstance(node.ctx, ast.Store)
            and node.attr
            in {"__cause__", "__context__", "__traceback__", "__suppress_context__"}
        ):
            result.add(definition_location(file["path"], node, node.attr))
    for node in ast.walk(file["tree"]):
        if (
            isinstance(node, ast.Attribute)
            and isinstance(node.ctx, ast.Store)
            and isinstance(node.value, ast.Name)
            and node.value.id in zip_infos
        ):
            result.add(definition_location(file["path"], node, node.attr))
    return result


def read_files(items):
    root = os.path.realpath(os.getcwd())
    files = []
    source_bytes = 0
    ast_nodes = 0
    for index, item in enumerate(items):
        exact_object(item, {"path", "module", "package"}, f"file {index}")
        path = exact_path(item["path"])
        if item["module"] and not identifier_chain(item["module"]):
            raise ValueError(f"file {index} has an invalid module")
        if item["package"] and not identifier_chain(item["package"]):
            raise ValueError(f"file {index} has an invalid package")
        target = contained_file(root, path)
        with open(target, "rb") as source_file:
            data = source_file.read(4 * 1024 * 1024 + 1)
        if len(data) > 4 * 1024 * 1024:
            raise ValueError(f"Python source {path} exceeds 4 MiB")
        source = data.decode("utf-8")
        tree = ast.parse(source, filename=path)
        source_bytes += len(data)
        ast_nodes += sum(1 for _ in ast.walk(tree))
        if source_bytes > MAX_SOURCE_BYTES or ast_nodes > MAX_AST_NODES:
            raise ValueError(
                "Python project exceeds its source or AST resource boundary"
            )
        files.append(
            {
                "path": path,
                "module": item["module"],
                "package": item["package"],
                "source": source,
                "tree": tree,
            }
        )
    paths = [file["path"] for file in files]
    if paths != sorted(set(paths)):
        raise ValueError("request Python paths must be unique and sorted")
    return files


def exact_text(value):
    return (
        isinstance(value, str)
        and value
        and "\x00" not in value
        and len(value.encode("utf-8")) <= MAX_STRING_BYTES
    )


def exact_contract_names(value, qualified=False):
    return (
        isinstance(value, list)
        and len(value) <= MAX_CONTRACT_NAMES
        and value == sorted(set(value))
        and all(
            identifier_chain(name) and (qualified or "." not in name) for name in value
        )
    )


def exact_contract(contract, index, identifiers):
    exact_object(
        contract,
        {
            "id",
            "kind",
            "target",
            "members",
            "attributes",
            "decorators",
            "annotatedFields",
            "keywords",
        },
        f"contract {index}",
    )
    keywords = contract["keywords"]
    valid = (
        exact_text(contract["id"])
        and contract["id"] not in identifiers
        and contract["kind"] in {"entry-point", "decorator", "module-binding", "type"}
        and exact_text(contract["target"])
        and exact_contract_names(contract["members"])
        and exact_contract_names(contract["attributes"])
        and exact_contract_names(contract["decorators"], qualified=True)
        and type(contract["annotatedFields"]) is bool
        and isinstance(keywords, dict)
        and len(keywords) <= MAX_CONTRACT_KEYWORDS
        and all(
            identifier_chain(name) and "." not in name and type(value) is bool
            for name, value in keywords.items()
        )
    )
    if contract["kind"] == "entry-point":
        parts = contract["target"].split(":")
        valid = (
            valid and len(parts) == 2 and all(identifier_chain(part) for part in parts)
        )
        valid = valid and not contract["attributes"] and not contract["decorators"]
        valid = valid and not contract["annotatedFields"] and not keywords
    elif contract["kind"] == "decorator":
        valid = valid and identifier_chain(contract["target"])
        valid = valid and not contract["members"] and not contract["attributes"]
        valid = valid and not contract["decorators"] and not contract["annotatedFields"]
    elif contract["kind"] == "module-binding":
        valid = valid and identifier_chain(contract["target"])
        valid = valid and bool(contract["members"]) and not contract["attributes"]
        valid = valid and not contract["decorators"] and not contract["annotatedFields"]
        valid = valid and not keywords
    else:
        valid = valid and identifier_chain(contract["target"])
        valid = valid and (
            bool(contract["members"])
            or bool(contract["attributes"])
            or contract["annotatedFields"]
            or bool(contract["decorators"])
        )
        valid = valid and not keywords
    if not valid:
        raise ValueError(f"contract {index} is invalid")
    identifiers.add(contract["id"])


def reachability_inputs(request):
    references = request["references"]
    backends = request["backends"]
    contracts = request["contracts"]
    if (
        not isinstance(references, list)
        or not isinstance(backends, list)
        or not isinstance(contracts, list)
        or len(references) + len(backends) + len(contracts) > MAX_REFERENCES
    ):
        raise ValueError("request has an invalid reachability count")
    identifiers = set()
    for index, reference in enumerate(references):
        exact_object(
            reference,
            {"id", "module", "symbol", "members"},
            f"reference {index}",
        )
        if (
            not exact_text(reference["id"])
            or reference["id"] in identifiers
            or not identifier_chain(reference["module"])
            or not identifier_chain(reference["symbol"])
            or not isinstance(reference["members"], list)
            or len(reference["members"]) > MAX_REFERENCES
            or any(
                not identifier_chain(member) or "." in member
                for member in reference["members"]
            )
            or reference["members"] != sorted(set(reference["members"]))
        ):
            raise ValueError(f"reference {index} is invalid")
        identifiers.add(reference["id"])
    for index, backend in enumerate(backends):
        exact_object(backend, {"id", "module", "object"}, f"backend {index}")
        if (
            not exact_text(backend["id"])
            or backend["id"] in identifiers
            or not identifier_chain(backend["module"])
            or backend["object"]
            and not identifier_chain(backend["object"])
        ):
            raise ValueError(f"backend {index} is invalid")
        identifiers.add(backend["id"])
    for index, contract in enumerate(contracts):
        exact_contract(contract, index, identifiers)
    return references, backends, contracts


def definition_chain(file, parts):
    body = file["tree"].body
    locations = []
    node = None
    for index, name in enumerate(parts):
        found = definition_sites(file, body, name)
        if len(found) != 1 or found[0][0] != "definition":
            raise ValueError("build backend object is stale or ambiguous")
        node = found[0][1]
        locations.append(definition_location(file["path"], node, name))
        if index < len(parts) - 1:
            if not isinstance(node, ast.ClassDef):
                raise ValueError("build backend object cannot resolve statically")
            body = node.body
    return locations, node


def backend_locations(modules, backend):
    file = module_file(modules, backend["module"])
    body = file["tree"].body
    result = []
    if backend["object"]:
        locations, node = definition_chain(file, backend["object"].split("."))
        result.extend(locations)
        if not isinstance(node, ast.ClassDef):
            raise ValueError("build backend object cannot expose hooks statically")
        body = node.body
    hooks = {
        "build_wheel",
        "build_sdist",
        "get_requires_for_build_wheel",
        "prepare_metadata_for_build_wheel",
        "get_requires_for_build_sdist",
        "build_editable",
        "get_requires_for_build_editable",
        "prepare_metadata_for_build_editable",
    }
    for name in sorted(hooks):
        found = definition_sites(file, body, name)
        if len(found) > 1:
            raise ValueError("build backend hook is ambiguous")
        if not found:
            continue
        match = found[0]
        if match[0] == "definition":
            result.append(definition_location(file["path"], match[1], name))
        else:
            node, item = match[1:]
            module = import_from_module(file, node)
            if not module:
                raise ValueError("build backend hook re-export has no module")
            result.extend(
                module_reference(modules, module, item.name.split("."), set())
            )
    return result


def reachability_locations(files, references, backends, contracts):
    modules = {}
    for file in files:
        if file["module"] and file["path"].endswith(".py"):
            modules.setdefault(file["module"], []).append(file)
    keep = set()
    problems = []
    semantic_contracts = [
        contract
        for contract in contracts
        if contract["kind"] in {"entry-point", "type"}
    ]
    context = (
        ContainedAstroidReferences(files, contracts)
        if semantic_contracts
        else nullcontext()
    )
    with context as resolver:
        for reference in references:
            try:
                parts = reference["symbol"].split(".")
                locations = module_reference(modules, reference["module"], parts, set())
                for member in reference["members"]:
                    locations.extend(
                        module_reference(
                            modules,
                            reference["module"],
                            parts + [member],
                            set(),
                        )
                    )
                keep.update(locations)
            except (TypeError, ValueError) as error:
                problems.append({"id": reference["id"], "message": bounded(str(error))})
        for contract in contracts:
            try:
                if contract["kind"] == "entry-point":
                    locations = resolver.entry_point(contract)
                elif contract["kind"] == "type":
                    locations = type_contract_locations(files, contract, resolver)
                elif contract["kind"] == "decorator":
                    locations = decorator_locations(files, contract)
                else:
                    locations = module_binding_locations(files, contract)
                if not locations:
                    raise ValueError("contract matches no source definitions")
                keep.update(locations)
            except (TypeError, ValueError) as error:
                problems.append({"id": contract["id"], "message": bounded(str(error))})
    for backend in backends:
        try:
            keep.update(backend_locations(modules, backend))
        except (TypeError, ValueError) as error:
            problems.append({"id": backend["id"], "message": bounded(str(error))})
    problems.sort(key=lambda item: item["id"])
    return keep, problems


def analyze(request):
    exact_object(
        request,
        {
            "protocol",
            "toolVersion",
            "files",
            "references",
            "backends",
            "contracts",
        },
        "request",
    )
    if request["protocol"] != PROTOCOL or request["toolVersion"] != TOOL_VERSION:
        raise ValueError("request has the wrong protocol or tool version")
    if (
        not isinstance(request["files"], list)
        or not 1 <= len(request["files"]) <= MAX_FILES
    ):
        raise ValueError("request has an invalid Python source count")
    references, backends, contracts = reachability_inputs(request)
    if sys.version_info[:2] != (3, 12):
        raise ValueError("Python dead-code analysis requires CPython 3.12")
    installed_version = importlib.metadata.version("vulture")
    if installed_version != TOOL_VERSION:
        raise ValueError(f"Python dead-code analysis requires Vulture {TOOL_VERSION}")
    if any(contract["kind"] in {"entry-point", "type"} for contract in contracts):
        installed_astroid = importlib.metadata.version("astroid")
        if installed_astroid != ASTROID_VERSION:
            raise ValueError(
                f"Python contract analysis requires Astroid {ASTROID_VERSION}"
            )
    files = read_files(request["files"])
    paths = [file["path"] for file in files]
    import vulture.core as vulture_core

    vulture_core.noqa.parse_noqa = lambda _: defaultdict(set)
    analyzer = vulture_core.Vulture()
    for file in files:
        analyzer.scan(file["source"], filename=file["path"])
    for name in {item.name for item in analyzer.defined_imports}:
        whitelist = f"whitelists/{name}_whitelist.py"
        try:
            data = pkgutil.get_data("vulture", whitelist)
        except OSError:
            continue
        if data is not None:
            analyzer.scan(data.decode("utf-8"), filename=whitelist)
    keep = set()
    for file in files:
        keep.update(callback_locations(file))
    reachability, problems = reachability_locations(
        files, references, backends, contracts
    )
    keep.update(reachability)
    if problems:
        return [], problems, paths
    dead_code = []
    for item in analyzer.get_unused_code(min_confidence=60):
        identity = (
            str(item.filename),
            item.first_lineno,
            item.last_lineno,
            item.name,
        )
        if identity[0] not in paths or identity in keep:
            continue
        if (
            not item.name
            or len(item.name.encode("utf-8")) > MAX_STRING_BYTES
            or item.typ not in KINDS
            or type(item.confidence) is not int
            or not 60 <= item.confidence <= 100
            or not item.message
            or len(item.message.encode("utf-8")) > MAX_STRING_BYTES
        ):
            raise ValueError("Vulture produced an invalid dead-code fact")
        dead_code.append(
            {
                "analyzer": "vulture",
                "path": identity[0],
                "line": identity[1],
                "endLine": identity[2],
                "name": identity[3],
                "kind": item.typ,
                "confidence": item.confidence,
                "message": item.message,
            }
        )
    if len(dead_code) > MAX_FACTS:
        raise ValueError("Vulture produced more than 10000 dead-code facts")
    dead_code.sort(
        key=lambda item: (
            item["path"],
            item["line"],
            item["endLine"],
            item["name"],
            item["kind"],
            item["confidence"],
            item["message"],
        )
    )
    return dead_code, problems, paths


def response(failure="", covered=None, dead_code=None, problems=None):
    return {
        "protocol": PROTOCOL,
        "toolVersion": TOOL_VERSION,
        "covered": [] if covered is None else covered,
        "deadCode": [] if dead_code is None else dead_code,
        "problems": [] if problems is None else problems,
        "failure": bounded(failure),
    }


def main():
    try:
        data = sys.stdin.buffer.read(MAX_INPUT_BYTES + 1)
        if len(data) > MAX_INPUT_BYTES:
            raise ValueError("request exceeds the byte limit")
        request = json.loads(data)
        dead_code, problems, covered = analyze(request)
        result = response(covered=covered, dead_code=dead_code, problems=problems)
    except (
        ImportError,
        OSError,
        SyntaxError,
        TypeError,
        UnicodeError,
        ValueError,
    ) as error:
        result = response(str(error))
    output = json.dumps(
        result, ensure_ascii=False, separators=(",", ":"), allow_nan=False
    ).encode("utf-8")
    if len(output) > MAX_OUTPUT_BYTES:
        output = json.dumps(
            response("response exceeds the byte limit"), separators=(",", ":")
        ).encode("utf-8")
    sys.stdout.buffer.write(output + b"\n")


main()
